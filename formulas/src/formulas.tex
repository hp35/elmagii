\eqsection{Konstitutiva relationer}
\sidx{Konstitutiva relationer}
\eqtitle{Elektrisk polarisationsdensitet}
\sidx{Elektrisk polarisationsdensitet ${\bf P}$}
$$
  {\bf P} \equiv \Big\langle{{d{\bf p}}\over{dV}}\Big\rangle
    = \varepsilon_0\chi_{\rm e}{\bf E},\qquad
  \nabla\cdot{\bf P}=-\rho_{\rm b}
$$
\eqtitle{Elektrisk fl{\"o}dest{\"a}thet}
\sidx{Elektrisk fl{\"o}dest{\"a}thet ${\bf D}$}
$$
  {\bf D}\equiv\varepsilon_0{\bf E}+{\bf P}
    =\varepsilon_0\varepsilon_{\rm r}{\bf E}
$$
\eqtitle{Magnetisering}
\sidx{Magnetisering ${\bf M}$}
$$
  {\bf M} \equiv \Big\langle{{d{\bf m}}\over{dV}}\Big\rangle
    = {{1}\over{\mu_0}}\Big(1-{{1}\over{\mu_{\rm r}}}\Big){\bf B}
$$
\eqtitle{Magnetisk fl{\"o}dest{\"a}thet}
\sidx{Magnetisk fl{\"o}dest{\"a}thet ${\bf B}$}
$$
  {\bf B}\equiv\mu_0({\bf H}+{\bf M})
    =\mu_0\mu_{\rm r}{\bf H}
$$

\eqsection{Elektrostatik}
\sidx{Elektrostatik}
%\eqsection{Punktladdningar}
\sidx{Punktladdningar}
\eqtitle{System av $N$ laddningar $q_k$ placerade i k{\"a}llpunkter ${\bf x}_k$}
$$
  \rho({\bf x})=\sum^{N}_{k=1}q_k\delta({\bf x}-{\bf x}'_k)\,dV
$$
\eqtitle{Coulombs kraftlag}
\sidx{Coulombs kraftlag}
$$
  {\bf F}({\bf x})={{qq'}\over{4\pi\varepsilon_0}}
     {{({\bf x}-{\bf x}')}\over{|{\bf x}-{\bf x}'|^3}}
     =q{\bf E}({\bf x})
$$
\eqtitle{Elektrisk f{\"a}ltstyrka fr{\aa}n punktladdning $q'$}
\sidx{Elektrisk f{\"a}ltstyrka ${\bf E}$}
$$
  {\bf E}({\bf x})={{q'}\over{4\pi\varepsilon_0}}
    {{({\bf x}-{\bf x}')}\over{|{\bf x}-{\bf x}'|^3}}
      ={{q'}\over{4\pi\varepsilon_0 r^2}}{\bf e}_r
$$
\eqtitle{Coulombs generaliserade lag -- ``Coulombintegralen''}
\sidx{Coulombs generaliserade lag}[Coulombintegralen]
$$
  {\bf E}({\bf x})={{1}\over{4\pi\varepsilon_0}}\iiint_V\rho({\bf x}')
    {{({\bf x}-{\bf x}')}\over{|{\bf x}-{\bf x}'|^3}}\,dV'
$$
\eqtitle{Existens av skal{\"a}r elektrostatisk potential}
\sidx{Skal{\"a}r potential $\phi$}[Elektrostatisk]
\sidx{Skal{\"a}r potential $\phi$}[Existens f{\"o}r]
\vskip2pt
Ur Coulombs generaliserade lag f{\"o}r elektrostatiska f{\"a}lt erh{\aa}lls
$\nabla\times{\bf E}={\bf 0}$, vilket visar att den elektriska f{\"a}ltstyrkan
alltid kan skrivas som en gradient av en skal{\"a}r funktion, den skal{\"a}ra
potentialen $\phi$, definierad som
$$
  {\bf E}=-\nabla\phi.
$$
\eqtitle{Elektriskt fl{\"o}de}
\sidx{Elektrisk fl{\"o}dest{\"a}thet ${\bf D}$}
$$
  \Phi_{\rm E}=\oiint_S {\bf E}({\bf x})\cdot d{\bf S}
     ={{1}\over{\varepsilon_0}}\iiint_V\rho({\bf x})\,dV
     ={{q_{\rm tot}}\over{\varepsilon_0}}
$$
\eqtitle{Gauss lag f{\"o}r elektrisk fl{\"o}dest{\"a}thet}
\sidx{Gauss lag}[F{\"o}r elektrisk fl{\"o}dest{\"a}thet ${\bf D}$]
$$
  \oiint{\bf D}\cdot d{\bf S}=\iiint\rho\,dV
$$
\eqtitle{Poissons ekvation (Coulombs lag) f{\"o}r skal{\"a}r potential}
\sidx{Poissons ekvation}[F{\"or} skal{\"a}r potential $\phi$]
$$
  \nabla^2\phi({\bf x})=-\rho({\bf x})/\varepsilon_0
$$
\eqtitle{Skal{\"a}r potential, explicit l{\"o}sning till Poissons ekvation}
\sidx{Skal{\"a}r potential $\phi$}[Explicit l{\"o}sning till Poissons ekvation]
$$
  \phi({\bf x})\equiv{{1}\over{4\pi\varepsilon_0}}\iiint_V
      {{\rho({\bf x}')}\over{|{\bf x}-{\bf x}'|}}\,dV'
  %  \ \Rightarrow\ {\bf E}({\bf x})=-\nabla\phi({\bf x})
$$
\eqtitle{Skal{\"a}r elektrostatisk potential fr{\aa}n punktladdning $q'$}
\sidx{Skal{\"a}r potential $\phi$}[Elektrostatisk potential]
\sidx{Skal{\"a}r potential $\phi$}[Fr{\aa}n punktladdning $q'$]
$$
  \phi({\bf x})
    ={{1}\over{4\pi\varepsilon_0}}{{q'}\over{|{\bf x}-{\bf x}'|}},
      \quad{\bf x}\ne{\bf x}'.
$$
\eqtitle{Arbete som tillf{\"o}rs punktladdning $q$ vid f{\"o}rflyttning fr{\aa}n
         $a$ till $b$}
\sidx{Arbete och upplagrad energi}[Elektrostatisk laddning]
\sidx{Arbete och upplagrad energi}[Skal{\"a}r potential $\phi$]
$$
  W_{\rm e}=-\int^{{\bf x}_b}_{{\bf x}_a}{\bf F}({\bf x})\cdot d{\bf l}
          =q\big(\phi({{\bf x}_b})-\phi({{\bf x}_a})\big) % =W_b-W_a
$$

\eqsection{Magnetostatik}
\sidx{Magnetostatik}
\eqtitle{Lorentz-kraften p{\aa} fri laddning $q$ med hastighet ${\bf v}$}
\sidx{Lorentz-kraften ${\bf F}_{\rm m}=q\,{\bf v}\times{\bf B}$}
$$
  {\bf F}=q\big({\bf E}+{\bf v}\times{\bf B}\big)
$$
\eqtitle{Amp\`eres kraftlag p{\aa} str{\"o}mslinga b{\"a}rande
  str{\"o}mmen $I$}
\sidx{Amp{\`e}res kraftlag}
$$
  {\bf F}_{\rm mag}=\int^{{\bf x}_b}_{{\bf x}_a} ({\bf I}\times{\bf B})\,dl
$$
\eqtitle{Str{\"o}mmen $I$ genom en yta $S$ med str{\"o}mt{\"a}thet}
\sidx{Elektrisk str{\"o}m $I$}
\sidx{Elektrisk str{\"o}mt{\"a}thet ${\bf J}$}
$$
  I=\iint_S {\bf J}\cdot d{\bf S}
$$
\eqtitle{Str{\"o}mt{\"a}thet i medium med konduktivitet $\sigma$}
\sidx{Str{\"o}mt{\"a}thet ${\bf J}$}[I medium med konduktivitet $\sigma$]
\sidx{Elektrisk konduktivitet $\sigma$}
$$
  {\bf J}=\sigma{\bf E}
$$
\eqtitle{Lagen om att elektrisk laddning inte kan f{\"o}rsvinna --
         kontinuitetsekvationen}
\sidx{Kontinuitetsekvationen}
$$
  \nabla\cdot{\bf J}=-{{d\rho}\over{dt}}.
$$
\eqtitle{Statiska problem}
\sidx{Elektrostatik}[Krav f{\"o}r statiska f{\"a}lt]
\sidx{Magnetostatik}[Krav f{\"o}r statiska f{\"a}lt]
$$
  {{d\rho}\over{dt}}=0\quad\underline{\hbox{och}}\quad{{d{\bf J}}\over{dt}}=0.
$$
\eqtitle{Divergens av str{\"o}mt{\"a}theten i statiska problem}
\sidx{Str{\"o}mt{\"a}thet ${\bf J}$}[Divergens i statiska problem]
$$
  {{d\rho}\over{dt}}=0\quad\Leftrightarrow\quad\nabla\cdot{\bf J}=0
$$
\eqtitle{Biot--Savarts lag f{\"o}r str{\"o}mslingor}
\sidx{Biot--Savarts lag}[F{\"o}r str{\"o}mslinga]
$$
  {\bf B}({\bf x})={{\mu_0}\over{4\pi}}\int^{{\bf x}_b}_{{\bf x}_a}
      {{{\bf I}({\bf x}')\times({\bf x}-{\bf x}')}
        \over{|{\bf x}-{\bf x}'|^3}}\,dl'
$$
\eqtitle{Biot--Savarts generella lag}
\sidx{Biot--Savarts lag}
$$
  {\bf B}({\bf x})={{\mu_0}\over{4\pi}}\iiint_V
      {{{\bf J}({\bf x}')\times({\bf x}-{\bf x}')}
        \over{|{\bf x}-{\bf x}'|^3}}\,dV'
$$
\eqtitle{Existens av magnetostatisk vektorpotential}
\sidx{Vektorpotential ${\bf A}$}[Magnetostatisk]
Ur Biot--Savarts generella lag f{\"o}r magnetostatiska f{\"a}lt erh{\aa}lls
$\nabla\cdot{\bf B}={\bf 0}$, vilket visar att den magnetiska
fl{\"o}dest{\"a}theten alltid kan skrivas som en rotation av en vektorfunktion,
vektorpotentialen ${\bf A}$, definierad som
$$
  {\bf B}=\nabla\times{\bf A}.
$$
\eqtitle{Gauss lag f{\"o}r magnetisk fl{\"o}dest{\"a}thet --
         icke-existens f{\"o}r magnetiska mono\-poler} % existerar inte}
\sidx{Gauss lag}[F{\"o}r magnetisk fl{\"o}dest{\"a}thet ${\bf B}$]
\sidx{Magnetisk fl{\"o}dest{\"a}thet ${\bf B}$}[Gauss lag]
\sidx{Magnetisk monopol}
$$
  \nabla\cdot{\bf B}=0\qquad\hbox{(alltid)}
$$
\eqtitle{Amp\`eres statiska lag}
\sidx{Amp\`eres lag}[F{\"o}r statiska magnetiska f{\"a}lt]
$$
  \oint_{\Gamma}{\bf B}\cdot d{\bf l}
    =\mu_0\iint_S{\bf J}\cdot d{\bf S}
    =\mu_0I_{\rm enc}
  \quad\Leftrightarrow\quad
  \nabla\times{\bf B}=\mu_0{\bf J}
$$
\eqtitle{Bunden elektrisk str{\"o}mt{\"a}thet}
\sidx{Elektrisk str{\"o}mt{\"a}thet ${\bf J}$}[Bunden]
$$
  {\bf J}_{\rm b}=\nabla\times{\bf M}
$$
\eqtitle{Bunden ytstr{\"o}mt{\"a}thet}
\sidx{Ytstr{\"o}mt{\"a}thet ${\bf K}$}[Bunden,
  ${\bf K}_{\rm b}={\bf M}\times{\bf e}_n$]
$$
  {\bf K}_{\rm b}={\bf M}\times{\bf e}_n
$$
\eqtitle{Poissons ekvation (Amp\`eres lag) f{\"o}r vektorpotentialen}
\sidx{Poissons ekvation}
\sidx{Poissons ekvation}[F{\"o}r vektorpotentialen ${\bf A}$]
$$
  \nabla^2{\bf A}=-\mu_0{\bf J},
$$
\eqtitle{Vektorpotential, explicit l{\"o}sning till Poissons ekvation}
\sidx{Vektorpotential ${\bf A}$}[Explicit l{\"o}sning till Poissons ekvation]
$$
  {\bf A}({\bf x})={{\mu_0}\over{4\pi}}\iiint_V
    {{{\bf J}({\bf x}')}\over{|{\bf x}-{\bf x}'|}}\,dV'
$$

${\bf x}'$ och f{\"a}ltpunkt ${\bf x}$ ser ut, som\par

    \sidx{Monopol}$\phi({\bf x})\sim 1/r$}\hfil}
\line{\hbox to 20pt{\hfil}{\hbox to110pt{Dipol (2-pol):\hfil}
    \sidx{Dipol}$\phi({\bf x})\sim 1/r^2$}\hfil}
\line{\hbox to 20pt{\hfil}{\hbox to110pt{Kvadrupol (4-pol):\hfil}
    \sidx{Kvadrupol}$\phi({\bf x})\sim 1/r^3$}\hfil}
\line{\hbox to 20pt{\hfil}{\hbox to110pt{Oktopol (8-pol):\hfil}
    \sidx{Oktopol}$\phi({\bf x})\sim 1/r^4$}\hfil}
\line{\hbox to 20pt{\hfil}{\hbox to110pt{Hexadecapol (16-pol):\hfil}
    \sidx{Hexadecapol}$\phi({\bf x})\sim 1/r^5$}\hfil}

    \sidx{Dotriacontapol}$\phi({\bf x})\sim 1/r^6$}\hfil}

\eqtitle{Maclaurin-utveckling av laddnings- och str{\"o}mt{\"a}thet}
\sidx{Maclaurin-utveckling}[Av laddningst{\"a}thet $\phi$]
\sidx{Maclaurin-utveckling}[Av str{\"o}mt{\"a}thet ${\bf J}$]
\sidx{Skal{\"a}r potential $\phi$}
\sidx{Vektorpotential ${\bf A}$}
$$
  \eqalign{
    \rho({\bf x}') &= \rho({\bf 0})
    +\sum^{3}_{k=1} x'_k
       {{\partial \rho({\bf x}')}
         \over{\partial x'_k}}\bigg|_{{\bf x}'={\bf 0}}\cr
    &\hskip50pt
    +{{1}\over{2}}\sum^{3}_{j=1}\sum^{3}_{k=1} x'_j x'_k
       {{\partial^2 \rho({\bf x}')}
         \over{\partial x'_j\partial x'_k}}\bigg|_{{\bf x}'={\bf 0}}
    +\ldots\cr
    {\bf J}({\bf x}') &= {\bf J}({\bf 0})
    +\sum^{3}_{k=1} x'_k
       {{\partial {\bf J}({\bf x}')}
         \over{\partial x'_k}}\bigg|_{{\bf x}'={\bf 0}}\cr
    &\hskip50pt
    +{{1}\over{2}}\sum^{3}_{j=1}\sum^{3}_{k=1} x'_j x'_k
       {{\partial^2 {\bf J}({\bf x}')}
         \over{\partial x'_j\partial x'_k}}\bigg|_{{\bf x}'={\bf 0}}
    +\ldots\cr
  }
$$

\eqtitle{Multipolmoment f{\"o}r system av punktladdningar}
\sidx{Multipolutveckling}[System av punktladdningar]
F{\"o}r ett system av $N$ punktladdningar $q'_k$ placerade vid k{\"a}llpunkterna
${\bf x}'_k$ {\"a}r den statiska multipolutvecklingen av skal{\"a}ra potentialen
$$
  \eqalign{
    \phi({\bf x})
      &={{1}\over{4\pi\varepsilon_0}}\iiint_{{\Bbb R}^3}
         {{\rho({\bf x}')}\over{|{\bf x}-{\bf x}'|}}\,dV'
       ={{1}\over{4\pi\varepsilon_0}}\sum^N_{k=1}
         {{q'_k}\over{|{\bf x}-{\bf x}'_k|}}.\cr
  }
$$

\eqtitle{Multipolmoment f{\"o}r kontinuum av laddning}
\sidx{Multipolutveckling}[Generell av laddningst{\"a}thet $\rho$]
F{\"o}r en kontinuerlig statisk laddningst{\"a}thet $\rho({\bf x})$ {\"a}r den
statiska multipolutvecklingen av skal{\"a}ra potentialen
$$
  \eqalign{
    \phi({\bf x})
      &={{1}\over{4\pi\varepsilon_0}}
        \bigg(
        {{1}\over{|{\bf x}|}}q
      +\sum^{3}_{k=1} {{x_k}\over{|{\bf x}|^3}}p_k
      \cr&\hskip26pt
      +{{1}\over{2}}\sum^{3}_{j=1}\sum^{3}_{k=1}
          {{(3 x_j x_k - \delta_{jk}|{\bf x}|^2)}\over{|{\bf x}|^5}}Q_{jk}
      +\ldots
        \bigg),
  }
$$
d{\"a}r\par
\line{%
  {\hbox to14pt{\hfil}}%
  {\hbox to120pt{%
    $\displaystyle q=\iiint_{V} \rho({\bf x}')\,dV'$,\hfil}}%
  {\hbox to98pt{(elektrisk monopol)\hfil}}}
\line{%
  {\hbox to14pt{\hfil}}%
  {\hbox to120pt{%
    $\displaystyle p_k=\iiint_{V} x'_k \rho({\bf x}')\,dV'$,\hfil}}%
  {\hbox to98pt{(dipolmoment)\hfil}}}
\line{%

\eqtitle{Gauge-transformation}
\sidx{Gauge-transformation}
Med den godtyckliga och tv{\aa} g{\aa}nger i rum och tid kontinuerligt
differentierbara gauge-funktionen $\psi({\bf x},t)$, {\"a}r
gauge-transformationen definierad som
$$
  \eqalign{
    {\bf A}'({\bf x},t)
      &={\bf A}({\bf x},t)+\nabla\psi({\bf x},t),\cr
    \phi'({\bf x},t)
      &=\phi({\bf x},t)-{{\partial\psi({\bf x},t)}\over{\partial t}},\cr
  }
$$
vilken l{\"a}mnar de elektromagnetiska f{\"a}lten invarianta under
vektoridentiteten $\nabla\times\nabla\phi={\bf 0}$.

\eqsection{Lorenz-villkoret -- Lorenz gauge}
\eqtitle{Gauge-transformation under Lorenz-villkoret}
\sidx{Lorenz-villkoret}
\sidx{Lorenz-villkoret}[Gauge-transformation]
Under Lorenz-villkoret v{\"a}ljs gauge-funktionen $\psi$ s{\aa} att
$$
  \nabla^2\psi({\bf x},t)
    -{{1}\over{c^2}}{{\partial^2\psi({\bf x},t)}\over{\partial t^2}}
  =-\nabla\cdot{\bf A}({\bf x},t)
        -{{1}\over{c^2}}{{\partial\phi({\bf x},t)}\over{\partial t}},
$$
vilket leder till att
$$
  \nabla\cdot{\bf A}'({\bf x},t)+
    {{1}\over{c^2}}{{\partial\phi'({\bf x},t)}\over{\partial t}} = 0.
$$
Lorenz-villkoret {\"a}r uppkallat efter
den danske fysikern {\it Ludvig Lorenz} (1829--1891), inte Hendrik Antoon
Lorentz (1853--1928).
\sidx{Lorentz, Hendrik Antoon (1853--1928)}
\sidx{Lorenz, Ludvig (1829--1891)}

\eqtitle{Lorenz-villkoret -- V{\aa}gekvationer f{\"o}r potentialer}
\sidx{Lorenz-villkoret}[V{\aa}gekvationer f{\"o}r potentialer]
V{\aa}gekvationerna frikopplas f{\"o}r den skal{\"a}ra potenti\-alen och
vektorpotentialen, till
$$
  \eqalign{
    \Big(
      \nabla^2-{{1}\over{c^2}}{{\partial^2}\over{\partial t^2}}
    \Big)\phi({\bf x},t)
        &=-{{\rho({\bf x},t)}\over{\varepsilon_0\varepsilon_{\rm r}}},\cr
    \Big(
      \nabla^2-{{1}\over{c^2}}{{\partial^2}\over{\partial t^2}}
    \Big){\bf A}({\bf x},t)
        &=-\mu_0{\bf J}_{\rm f}({\bf x},t),\cr
  }
$$
d{\"a}r v{\aa}gutbredningshastigheten f{\"o}r b{\aa}de skal{\"a}ra potentialen
och vektorpotentialen {\"a}r $c^2$, det vill s{\"a}ga att b{\aa}da potentialerna
uppvisar en explicit kausal v{\aa}gut\-bredning med ljushastigheten.

\eqtitle{Lorenz-villkoret -- Retarderade potentialer}
\sidx{Lorenz-villkoret}[Retarderade potentialer]
L{\"o}sningar till v{\aa}gekvationerna erh{\aa}lls som de retarde\-rade
potentialerna
$$
  \eqalign{
    \phi({\bf x},t)&={{1}\over{4\pi\varepsilon_0}}\iiint_{{\Bbb R}^3}
      {{\rho({\bf x}',t')}\over{|{\bf x}-{\bf x}'|}}\,dV',\cr
    {\bf A}({\bf x},t)&={{\mu_0}\over{4\pi}}\iiint_{{\Bbb R}^3}
      {{{\bf J}_{\rm f}({\bf x}',t')}\over{|{\bf x}-{\bf x}'|}}\,dV',\cr
  }
$$
d{\"a}r $t'=t-|{\bf x}-{\bf x}'|/c$ {\"a}r den {\it retarderade tiden}
fr{\aa}n ${\bf x}'$ till ${\bf x}$.

\eqsection{Coulomb-villkoret -- Coulomb gauge}
\eqtitle{Gauge-transformation under Coulomb-villkoret}
\sidx{de Coulomb, Charles-Augustin (1736--1806)}
\sidx{Coulomb-villkoret}[Gauge-transformation]
Under Coulomb-villkoret, {\"a}ven kallat {\it transverse gauge} eller
{\it radiation gauge}, v{\"a}ljs gauge-funktionen $\psi$ s{\aa} att
$$
  \nabla^2\psi({\bf x},t)
    -{{1}\over{c^2}}{{\partial^2\psi({\bf x},t)}\over{\partial t^2}}
  =-{{1}\over{c^2}}{{\partial\phi({\bf x},t)}\over{\partial t}},
$$
vilket leder till att
$$
  \nabla\cdot{\bf A}'({\bf x},t)=0.
$$

\eqtitle{Coulomb-villkoret -- V{\aa}gekvationer f{\"o}r potentialer}
\sidx{Coulomb-villkoret}[V{\aa}gekvationer f{\"o}r potentialer]
V{\aa}gekvationerna f{\"o}r den skal{\"a}ra potentialen och vektorpotentialen
antar formen
$$
  \eqalign{
    \nabla^2\phi({\bf x},t)
      &=-{{\rho({\bf x},t)}\over{\varepsilon_0\varepsilon_{\rm r}}},\cr
    \bigg(
       \nabla^2-{{1}\over{c^2}}{{\partial^2}\over{\partial t^2}}
     \bigg){\bf A}({\bf x},t)
      &=-\mu_0{\bf J}_{\rm f}({\bf x},t)
      \cr&\hskip30pt
        +{{1}\over{c^2}}\nabla{{\partial\phi({\bf x},t)}\over{\partial t}},\cr
  }
$$
d{\"a}r den skal{\"a}ra potentialen $\phi$ uppfyller den statiska Poissons
ekvation utan tidsderivata, medan vektorpotentialen ${\bf A}$ uppfyller
v{\aa}gekvationen med den skal{\"a}ra potentialen som extra k{\"a}llterm.

\eqtitle{Coulomb-villkoret -- Instantan skal{\"a}r potential}
\sidx{Coulomb-villkoret}[Instantan skal{\"a}r potential]
L{\"o}sningar till v{\aa}g\-ekva\-tion\-erna erh{\aa}lls som
$$
  \eqalign{
    \phi({\bf x},t)&={{1}\over{4\pi\varepsilon_0}}\iiint_{{\Bbb R}^3}
      {{\rho({\bf x}',t)}\over{|{\bf x}-{\bf x}'|}}\,dV',\cr
    {\bf A}({\bf x},t)&={{\mu_0}\over{4\pi}}\iiint_{{\Bbb R}^3}
      {{{\bf J}_{\rm T}({\bf x}',t')}\over{|{\bf x}-{\bf x}'|}}\,dV',\cr
  }
$$
d{\"a}r nu integralen f{\"o}r den skal{\"a}ra potentialen $\phi$ inte
l{\"a}ngre involverar n{\aa}gon retarderad tid, utan responderar momentant
vid den aktuella tiden $t$, men d{\"a}r integralen f{\"o}r vektorpotentialen
involverar den {\it retarderade tiden} $t'=t-|{\bf x}-{\bf x}'|/c$ fr{\aa}n
${\bf x}'$ till ${\bf x}$. H{\"a}r {\"a}r ${\bf J}_{\rm T}$ {\"a}r den
transversella str{\"o}mt{\"a}theten
$$
  {\bf J}_{\rm T}={\bf J}_{\rm f}
    -{{1}\over{4\pi}}\nabla\iiint_{{\Bbb R}^3}
     {{\nabla'\cdot{\bf J}_{\rm f}({\bf x}',t)}\over{|{\bf x}-{\bf x}'|}}\,dV'
$$
med $\nabla'$ opererande p{\aa} k{\"a}llkoordinaterna ${\bf x}'$.
